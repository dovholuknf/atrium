# Native `title` tooltips the board keeps on purpose. Read by scripts/check-titles.sh.
#
# One entry per line: `file:pattern # reason`. file is relative to internal/api/web. pattern is an extended regex
# matched against the line that sets the title. An entry with no reason gets removed.
#
# Nearly empty on purpose. Every icon-only control carries an aria-label for its name and a data-tip for the styled
# tooltip, so almost nothing on the board needs the browser's own box.

# A <select> option's own tooltip, which data-tip cannot reach.
js/settings.js:<option value= # native option tooltip
# The read page does not load tooltips.js, so data-tip would show nothing there.
read.html:class="zoom" # read page has no styled tooltips
# Third-party markdown renderer, emitting the title of a link a user wrote.
vendor/marked.js:title # not our markup
# The phone board loads no styled tooltips, so data-tip would show nothing there.
m/js/files.js:goes within a day # phone page has no styled tooltips
# A property of a plain data object (the work row's title), not an element's tooltip.
js/items-data.js:out\.title = wk\.title # data, not a tooltip
# The phone board loads no styled tooltips, so data-tip would show nothing there.
m/js/items.js:i\.title = s\.name # phone page has no styled tooltips
m/js/items.js:this card has ended and was removed # phone page has no styled tooltips
# The share page is standalone and loads no tooltips.js, so data-tip would show nothing there.
share.html:class="zoom" # share page has no styled tooltips
# The reader runs on read.html and share.html, neither of which loads tooltips.js.
read.js:a\.title = "not shared" # reader pages have no styled tooltips
